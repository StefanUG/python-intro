```python.run requires="codingpirates-maze"
from maze import Puzzle, BeePlayer

maze = Puzzle.from_file("courseE_bee_functions7")
bee: BeePlayer = maze.player

def get_7_make_7():
    bee.right()
    bee.forward()
    for i in range(7):
        bee.get_nectar()
    bee.forward()
    for i in range(7):
        bee.make_honey()
    bee.backward()
    bee.backward()
    bee.left()

bee.forward()
get_7_make_7()
for i in range(3):
    bee.forward()
get_7_make_7()

Puzzle.done()
```
