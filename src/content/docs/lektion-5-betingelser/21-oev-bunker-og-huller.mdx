```python.run requires="codingpirates-maze"
from maze import Puzzle, Farmer

maze = Puzzle.from_file("courseD_farmer_while10_2025")
farmer: Farmer = maze.player

for i in range(3):
    while farmer.path_ahead():
        farmer.forward()
    while farmer.at_pile():
        farmer.remove()
    farmer.left()
    while farmer.path_ahead():
        farmer.forward()
    farmer.right()

Puzzle.done()
```
